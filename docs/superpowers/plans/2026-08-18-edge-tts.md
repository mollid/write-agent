```python
"""edge-tts 封装：预设音色表、参数校验、合成到文件（重试 + 原子替换）。

设计要点（spec §3）：接入层隔离在此模块，后续可替换云端 TTS；
demo 模式的静音 MP3 分支由 Job 层决定，不在此判断。
"""
from __future__ import annotations

import asyncio
import os
import re
import time
from pathlib import Path

import edge_tts

TTS_VOICES: dict[str, str] = {
    "zh-CN-XiaoxiaoNeural": "晓晓（女 · 温暖通用）",
    "zh-CN-XiaoyiNeural": "晓伊（女 · 活泼）",
    "zh-CN-XiaochenNeural": "晓辰（女 · 知性）",
    "zh-CN-XiaomengNeural": "晓梦（女 · 温柔）",
    "zh-CN-XiaoqiuNeural": "晓秋（女 · 文艺）",
    "zh-CN-YunxiNeural": "云希（男 · 阳光）",
    "zh-CN-YunjianNeural": "云健（男 · 浑厚）",
    "zh-CN-YunyangNeural": "云扬（男 · 新闻）",
    "zh-CN-YunyeNeural": "云野（男 · 成熟）",
    "zh-CN-YunxiaNeural": "云夏（男 · 少年）",
}
DEFAULT_VOICE = "zh-CN-XiaoxiaoNeural"

_RATE_RE = re.compile(r"^[+-]\d+%$")
_VOLUME_RE = re.compile(r"^[+-]\d+%$")
_PITCH_RE = re.compile(r"^[+-]\d+Hz$")

# a few-KB valid silent MPEG frame; demo-mode artifact so the whole flow
# (job → file → audio player) is testable offline
SILENT_MP3 = (
    b"\xff\xfb\x90\x00" + b"\x00" * 413
) * 8  # 8 tiny frames ≈ 3.3 KB

_MAX_RETRIES = 2


def validate_tts_params(voice: str, rate: str, volume: str, pitch: str) -> None:
    if voice not in TTS_VOICES:
        raise ValueError(f"未知音色: {voice}")
    if not _RATE_RE.match(rate):
        raise ValueError(f"语速格式非法（应为 +20% 这样的百分比）: {rate}")
    if not _VOLUME_RE.match(volume):
        raise ValueError(f"音量格式非法（应为 -10% 这样的百分比）: {volume}")
    if not _PITCH_RE.match(pitch):
        raise ValueError(f"音调格式非法（应为 +5Hz 这样的赫兹值）: {pitch}")


def synth_to_file(text: str, voice: str, rate: str, volume: str, pitch: str,
                  out_path: Path) -> None:
    """Synthesize `text` to `out_path` (atomic replace); retry twice on failure.

    Runs in a Job thread: drive the asyncio API with asyncio.run per attempt.
    """
    validate_tts_params(voice, rate, volume, pitch)
    last_err: Exception | None = None
    for _attempt in range(_MAX_RETRIES + 1):
        try:
            tmp = out_path.with_suffix(out_path.suffix + ".tmp")

            async def _run() -> None:
                comm = edge_tts.Communicate(text, voice, rate=rate, volume=volume, pitch=pitch)
                await comm.save(str(tmp))

            asyncio.run(_run())
            os.replace(tmp, out_path)
            return
        except Exception as e:  # noqa: BLE001 - retried/surfaced to the job
            last_err = e
            time.sleep(2)
    assert last_err is not None
    raise last_err
```

注意：`import edge_tts` 放模块顶部（不在 Job 线程惰性导入也可接受——但为保持「tests import-free of network libs」约定，测试里 monkeypatch `novel.tts.edge_tts` 属性即可，模块顶部导入本身不联网）。

```python
def run_tts_job(proj_root: Path, chapter: int, voice: str, rate: str,
                volume: str, pitch: str) -> Callable[[Job], object]:
    """Chapter → MP3 via edge-tts (live) or a silent placeholder (demo).
    Emits stage:tts before the synth call so the JobStrip comes alive
    immediately, then tts_done."""
    from novel.runtime import is_live_mode
    from novel.tts import SILENT_MP3, synth_to_file

    def fn(job: Job) -> object:
        job.emit({"type": "stage", "stage": "tts", "chapter": chapter})
        text = (proj_root / "chapters" / f"{chapter:03d}.md").read_text(encoding="utf-8")
        audio_dir = proj_root / "audio"
        audio_dir.mkdir(parents=True, exist_ok=True)
        out = audio_dir / f"{chapter:03d}.mp3"
        if is_live_mode():
            synth_to_file(text, voice, rate, volume, pitch, out)
        else:
            tmp = out.with_suffix(out.suffix + ".tmp")
            tmp.write_bytes(SILENT_MP3)
            import os as _os
            _os.replace(tmp, out)
        job.emit({"type": "tts_done", "chapter": chapter})
        return {"chapter": chapter, "path": str(out)}

    return fn
```

    @app.post("/api/projects/{name}/tts", status_code=202)
    def start_tts(name: str, req: TtsReq) -> dict[str, str]:
        p = _safe_project(root, name)
        from novel.tts import validate_tts_params
        try:
            validate_tts_params(req.voice, req.rate, req.volume, req.pitch)
        except ValueError as e:
            raise HTTPException(status_code=422, detail=str(e)) from e
        if not (p / "chapters" / f"{req.chapter:03d}.md").is_file():
            raise HTTPException(status_code=404, detail=f"章节不存在: {req.chapter}")
        try:
            job = manager.start(name, "tts", jobs.run_tts_job(
                p, req.chapter, req.voice, req.rate, req.volume, req.pitch))
        except RuntimeError as e:
            raise HTTPException(status_code=409, detail=str(e)) from e
        return {"job_id": job.id}

    @app.get("/api/projects/{name}/audio")
    def list_audio(name: str) -> list[dict[str, int]]:
        p = _safe_project(root, name)
        audio_dir = p / "audio"
        if not audio_dir.is_dir():
            return []
        out = []
        for f in sorted(audio_dir.glob("*.mp3")):
            if f.stem.isdigit():
                out.append({"chapter": int(f.stem), "size": f.stat().st_size})
        return out

    @app.get("/api/projects/{name}/audio/{num}")
    def get_audio(name: str, num: int):
        p = _safe_project(root, name)
        f = p / "audio" / f"{num:03d}.mp3"
        if not f.is_file():
            raise HTTPException(status_code=404, detail=f"音频不存在: {num}")
        return FileResponse(f, media_type="audio/mpeg", filename=f.name)

    const hz = (v: number) => `${v >= 0 ? "+" : ""}${v}Hz`;
    window.localStorage.setItem(
      `tts-params:${name}`,
      JSON.stringify({ voice: ttsVoice, rate: ttsRate, volume: ttsVolume, pitch: ttsPitch }));
    setTtsPending(true);
    tts.mutate(
      { chapter: selected, voice: ttsVoice, rate: pct(ttsRate), volume: pct(ttsVolume), pitch: hz(ttsPitch) },
      {
        onSuccess: () => message.info("已开始合成，完成后可在此试听"),
        onError: (e) => { setTtsPending(false); message.error(String(e)); },
